def prepare_stock_list(context):
    context.no_trading_today_signal = today_is_between(context, 2, '01-01', '01-30')

    context.hold_list = []
    for position in list(context.portfolio.positions.values()):
        stock = position.security
        context.hold_list.append(stock)

    context.history_hold_list.append(context.hold_list)
    if len(context.history_hold_list) >= context.limit_days:
        context.history_hold_list = context.history_hold_list[-context.limit_days:]
    temp_set = set()
    for hold_list in context.history_hold_list:
        for stock in hold_list:
            temp_set.add(stock)
    context.not_buy_again_list = list(temp_set)

    if context.hold_list != []:
        df = get_price(context.hold_list, end_date=context.previous_date, frequency='daily', fields=['close', 'high_limit'], count=1, panel=False, fill_paused=False)
        df = df[df['close'] == df['high_limit']]
        context.high_limit_list = list(df.code)
    else:
        context.high_limit_list = []


def weekly_adjustment(context):
    if context.no_trading_today_signal or context.consistency_signal:
        close_account(context)
        logger.info('关仓期')
        return

    context.target_list = get_stock_list(context)
    context.target_list = filter_paused_stock(context.target_list)
    context.target_list = filter_limitup_stock(context, context.target_list)
    context.target_list = filter_limitdown_stock(context, context.target_list)

    for stock in context.hold_list:
        if (stock not in context.target_list) and (stock not in context.high_limit_list):
            logger.info("卖出[%s]" % (stock))
            position = context.portfolio.positions[stock]
            close_position(position)
        else:
            logger.info("已持有[%s]" % (stock))

    position_count = len(context.portfolio.positions)
    target_num = len(context.target_list)
    if target_num > position_count:
        value = context.portfolio.cash / (target_num - position_count)
        for stock in context.target_list:
            if context.portfolio.positions[stock].total_amount == 0:
                if open_position(stock, value):
                    if len(context.portfolio.positions) == target_num:
                        break


def check_limit_up(context):
    now_time = context.current_dt
    if context.high_limit_list != []:
        for stock in context.high_limit_list:
            current_data = get_price(stock, end_date=now_time, frequency='1m', fields=['close', 'high_limit'], skip_paused=False, fq='pre', count=1, panel=False, fill_paused=True)
            if current_data.iloc[0, 0] < current_data.iloc[0, 1]:
                logger.info("[%s]涨停打开，卖出" % (stock))
                position = context.portfolio.positions[stock]
                close_position(position)
            else:
                logger.info("[%s]涨停，继续持有" % (stock))


def check_high_volume(context):
    for stock in context.portfolio.positions:
        if context.portfolio.positions[stock].sellable_quantity == 0:
            continue
        df_volume = get_bars(stock, count=context.HV_duration, unit='1d', fields=['volume'], include_now=True, df=True)
        if df_volume['volume'].values[-1] > context.HV_ratio * df_volume['volume'].values.max():
            logger.info("[%s]天量，卖出" % stock)
            position = context.portfolio.positions[stock]
            close_position(position)


def today_is_between(context, date_flag, start_date, end_date):
    today = context.current_dt.strftime('%m-%d')
    if date_flag == 1:
        if (start_date <= today) and (today <= end_date):
            return True
        else:
            return False
    elif date_flag == 2:
        if ('01-01' <= today) and (today <= '01-31'):
            return True
        elif ('04-01' <= today) and (today <= '04-30'):
            return True
        else:
            return False
    elif date_flag == 3:
        if ('01-01' <= today) and (today <= '01-31'):
            return True
        elif ('04-01' <= today) and (today <= '04-30'):
            return True
        elif ('06-01' <= today) and (today <= '06-30'):
            return True
        else:
            return False
    return False


def close_account(context):
    if len(context.portfolio.positions) == 0:
        return
    for stock in context.portfolio.positions:
        position = context.portfolio.positions[stock]
        close_position(position)
        logger.info("关仓，卖出[%s]" % (stock))


def filter_paused_stock(stock_list):
    current_data = get_current_data()
    return [stock for stock in stock_list if not current_data[stock].paused]


def filter_st_stock(stock_list):
    current_data = get_current_data()
    return [stock for stock in stock_list
            if not current_data[stock].is_st
            and 'ST' not in current_data[stock].name
            and '*' not in current_data[stock].name
            and '退' not in current_data[stock].name]


def get_recent_limit_up_stock(context, stock_list, recent_days):
    stat_date = context.previous_date
    new_list = []
    for stock in stock_list:
        df = get_price(stock, end_date=stat_date, frequency='daily', fields=['close', 'high_limit'], count=recent_days, panel=False, fill_paused=False)
        df = df[df['close'] == df['high_limit']]
        if len(df) > 0:
            new_list.append(stock)
    return new_list


def filter_limitup_stock(context, stock_list):
    last_prices = history(1, unit='1m', field='close', security_list=stock_list)
    current_data = get_current_data()
    return [stock for stock in stock_list if stock in context.portfolio.positions.keys()
            or last_prices[stock][-1] < current_data[stock].high_limit]


def filter_limitdown_stock(context, stock_list):
    last_prices = history(1, unit='1m', field='close', security_list=stock_list)
    current_data = get_current_data()
    return [stock for stock in stock_list if stock in context.portfolio.positions.keys()
            or last_prices[stock][-1] > current_data[stock].low_limit]


def filter_kcb_stock(context, stock_list):
    return [stock for stock in stock_list if stock[0:3] != '688']


def filter_new_stock(context, stock_list, d):
    yesterday = context.previous_date
    return [stock for stock in stock_list if not yesterday - get_security_info(stock).start_date < datetime.timedelta(days=d)]


def order_target_value_(security, value):
    if value == 0:
        log.debug("Selling out %s" % (security))
    else:
        log.debug("Order %s to value %f" % (security, value))
    return order_target_value(security, value)


def open_position(security, value):
    order = order_target_value_(security, value)
    if order is not None and order.filled > 0:
        return True
    return False


def close_position(position):
    security = position.security
    order = order_target_value_(security, 0)
    if order is not None:
        if order.status == OrderStatus.held and order.filled == order.amount:
            return True
    return False


def adjust_position(context, buy_stocks, stock_num):
    for stock in context.portfolio.positions:
        if stock not in buy_stocks:
            logger.info("[%s]不在应买入列表中" % stock)
            position = context.portfolio.positions[stock]
            close_position(position)
        else:
            logger.info("[%s]已经持有无需重复买入" % stock)

    position_count = len(context.portfolio.positions)
    if stock_num > position_count:
        value = context.portfolio.cash / (stock_num - position_count)
        for stock in buy_stocks:
            if context.portfolio.positions[stock].total_amount == 0:
                if open_position(stock, value):
                    if len(context.portfolio.positions) == stock_num:
                        break
