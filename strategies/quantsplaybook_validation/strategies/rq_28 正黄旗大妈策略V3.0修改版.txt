def buystock(context, choice):
    position_count = len(context.portfolio.positions)
    if context.stock_num <= position_count:
        return

    buylist = choice
    cdata = get_current_data()
    psize = context.portfolio.cash / (context.stock_num - position_count)
    for s in buylist:
        if s not in context.portfolio.positions:
            logger.info('buy', s, cdata[s].name)
            order_value(s, psize)
            if len(context.portfolio.positions) == context.stock_num:
                break


def sellstock(context, sell_list):
    if len(sell_list) > 0:
        for security in sell_list:
            cprice = get_current_data()[security].last_price
            boughtcost = context.portfolio.positions[security].acc_avg_cost
            profit = (cprice - boughtcost) / boughtcost * 100
            logger.info("Sell %s " % get_current_data()[security].name, "profit: %.1f%%" % profit, "init time %s" % context.portfolio.positions[security].init_time)
            limit_price = max(cprice * 0.95, get_current_data()[security].low_limit)
            ordert = order_target_value(security, 0, LimitOrderStyle(limit_price))
            if ordert is None:
                logger.info("Sell failed %s" % get_current_data()[security].name)
    return


def prepare_stock_list(context):
    context.high_limit_list = []
    hold_list = list(context.portfolio.positions)
    if hold_list:
        df = get_price(hold_list, end_date=context.previous_date, frequency='daily',
                       fields=['close', 'high_limit'],
                       count=1, panel=False)
        context.high_limit_list = df[df['close'] == df['high_limit']]['code'].tolist()


def check_limit_up(context):
    cdata = get_current_data()
    sell_list = []
    if context.high_limit_list:
        for stock in context.high_limit_list:
            if cdata[stock].last_price < cdata[stock].high_limit:
                logger.info("[%s]涨停打开，卖出" % cdata[stock].name)
                sell_list.append(stock)
            else:
                logger.info("[%s]涨停，继续持有" % cdata[stock].name)
    if len(sell_list) > 0:
        sellstock(context, sell_list)


def check_at_dayend(context):

def filter_kcbj_stock(stock_list):
    for stock in stock_list[:]:
        if stock[0] == '4' or stock[0] == '8' or stock[:2] == '68':
            stock_list.remove(stock)
    return stock_list


def filter_paused_stock(stock_list):
    current_data = get_current_data()
    return [stock for stock in stock_list if not current_data[stock].paused]


def filter_st_stock(stock_list):
    current_data = get_current_data()
    return [stock for stock in stock_list
            if not current_data[stock].is_st
            and 'ST' not in current_data[stock].name
            and '*' not in current_data[stock].name
            and '退' not in current_data[stock].name]


def filter_limitup_stock(context, stock_list):
    last_prices = history(1, unit='1m', field='close', security_list=stock_list)
    current_data = get_current_data()
    return [stock for stock in stock_list if stock in context.portfolio.positions.keys()
            or last_prices[stock][-1] < current_data[stock].high_limit]


def filter_limitdown_stock(context, stock_list):
    last_prices = history(1, unit='1m', field='close', security_list=stock_list)
    current_data = get_current_data()
    return [stock for stock in stock_list if stock in context.portfolio.positions.keys()
            or last_prices[stock][-1] > current_data[stock].low_limit]


def filter_highprice_stock(context, stock_list):
    last_prices = history(1, unit='1m', field='close', security_list=stock_list)
    return [stock for stock in stock_list if stock in context.portfolio.positions.keys()
            or last_prices[stock][-1] < 9]
