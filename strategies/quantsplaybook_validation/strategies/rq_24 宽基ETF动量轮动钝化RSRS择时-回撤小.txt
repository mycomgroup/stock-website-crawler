def get_rank(context, stock_pool):
    score_list = []
    for stock in context.stock_pool:
        data = history_bars(stock, context.momentum_day, '1d', 'close')
        y = np.log(data)
        x = np.arange(len(y))
        slope, intercept = np.polyfit(x, y, 1)
        annualized_returns = math.exp(slope * 250) - 1
        r_squared = 1 - (sum((y - (slope * x + intercept)) ** 2) / ((len(y) - 1) * np.var(y, ddof=1)))
        score = annualized_returns * r_squared
        score_list.append(score)
    stock_dict = dict(zip(context.stock_pool, score_list))
    sort_list = sorted(stock_dict.items(), key=lambda item: item[1], reverse=True)
    code_list = [sort_list[i][0] for i in range(len(context.stock_pool))]
    rank_stock = code_list[0]
    return rank_stock


def get_ols(x, y):
    slope, intercept = np.polyfit(x, y, 1)
    r2 = 1 - (sum((y - (slope * x + intercept)) ** 2) / ((len(y) - 1) * np.var(y, ddof=1)))
    return intercept, slope, r2


def get_zscore(slope_series):
    mean = np.mean(slope_series)
    std = np.std(slope_series)
    return (slope_series[-1] - mean) / std


def get_timing_signal(context, stock):

    low_window = data['low'][-context.N[stock]:]
    high_window = data['high'][-context.N[stock]:]
    intercept, slope, r2 = get_ols(low_window, high_window)

    slope_series = [get_ols(data.low[i:i + context.N[stock]], data.high[i:i + context.N[stock]])[1] for i in range(context.M[stock])]
    slope_series.append(slope)
    zscore = get_zscore(slope_series[-context.M[stock]:])
    rsrs_score = zscore * r2 ** (2 * ret_quantile)

    if rsrs_score > context.score_threshold[stock]:
        return "BUY"
    elif rsrs_score < -context.score_threshold[stock]:
        return "SELL"
    else:
        return "KEEP"
