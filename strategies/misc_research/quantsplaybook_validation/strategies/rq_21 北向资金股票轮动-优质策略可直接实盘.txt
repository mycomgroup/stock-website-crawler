def etftrade(context):
    if context.signal == 'CLEAR':
        for stock in context.portfolio.positions:
            if stock == context.empty_keep_stock:
                continue
            logger.info("清仓: %s" % stock)
            order_target_value(stock, 0)
    elif context.signal == 'BUY':
        if context.empty_keep_stock in context.portfolio.positions:
            order_target_value(context.empty_keep_stock, 0)

        holdings = set(context.portfolio.positions.keys())
        targets = set(context.buy)

        sells = holdings - targets
        for code in sells:
            logger.info("卖出: %s" % code)
            order_target_value(code, 0)

        ratio = len(targets)
        if ratio > 0:
            cash = context.portfolio.total_value / ratio
            adjusts = holdings & targets
            for code in adjusts:
                if abs(cash - context.portfolio.positions[code].value) > 5000:
                    logger.info('调仓: %s' % code)
                    order_target_value(code, cash)

            purchases = targets - holdings
            for code in purchases:
                logger.info('买入: %s' % code)
                order_target_value(code, cash)

            current_returns = 100 * context.portfolio.returns
            logger.info("当前收益：%.2f%%，当前持仓: %s", current_returns, list(context.portfolio.positions.keys()))

    if len(context.portfolio.positions) == 0:
        order_target_value(context.empty_keep_stock, context.portfolio.cash)
