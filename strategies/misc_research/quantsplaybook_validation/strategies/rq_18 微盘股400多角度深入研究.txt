import pandas as pd


def init(context):
    # 设置系统
    set_log_level('order', 'error')
    set_option('use_real_price', True)
    set_option('avoid_future_data', True)
    context.days = 0


def after_code_changed(context):
    # 设置策略
    unschedule_all()
    run_daily(i_update, time_rule=market_open(minute=0))
    run_daily(i_trader, time_rule=market_open(minute=35))
    run_daily(i_report, time_rule=market_close())


def i_update(context):
    # 参数
    nchoice = 120
    nposition = 100
    # 全部A股
    dt_last = context.previous_date
    all_stock = all_instruments('CS')
    stocks = [s for s in all_stock.order_book_id if not get_current_data()[s].is_st]
    # 小盘股
    df = get_fundamentals(query(
            valuation.code,
            valuation.market_cap,
        ).filter(
            valuation.code.in_(stocks),
            valuation.pb_ratio > 0,
        ).order_by(
            valuation.market_cap.asc()
        ).limit(nchoice)
    ).dropna().set_index('code')
    # 结果
    context.choice = df.index.tolist()
    context.position_size = 1.0 / nposition * context.portfolio.total_value


def i_trader(context):
    # 参量
    choice = context.choice
    position_size = context.position_size
    lm_value = 0.8 * position_size
    hm_value = 1.2 * position_size
    cdata = get_current_data()
    # 卖出
    for s in context.portfolio.positions:
        if cdata[s].paused or \
            cdata[s].last_price >= cdata[s].high_limit or \
                cdata[s].last_price <= cdata[s].low_limit:
            continue  # 过滤三停
        if s not in choice:
            logger.info('sell', s, cdata[s].name)
            order_target_value(s, 0)
    # 买进
    for s in choice:
        if context.portfolio.cash < position_size:
            break
        if cdata[s].paused or \
            cdata[s].last_price >= cdata[s].high_limit or \
                cdata[s].last_price <= cdata[s].low_limit:
            continue  # 过滤三停
        if s not in context.portfolio.positions:
            logger.info('buy', s, cdata[s].name)
            order_target_value(s, position_size)
        elif context.portfolio.positions[s].value < lm_value:
            logger.info('balance+', s, cdata[s].name)
            order_target_value(s, position_size)
        elif context.portfolio.positions[s].value > hm_value:
            logger.info('balance-', s, cdata[s].name)
            order_target_value(s, position_size)
